local ALPHABET = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
local LOOKUP = {}
for i = 1, #ALPHABET do LOOKUP[ALPHABET:byte(i)] = i - 1 end

local function b64decode(s)
    s = s:gsub("[^%w%+/=]", "")
    local out, n = {}, 0
    local buf, bn = {}, 0
    for i = 1, #s, 4 do
        local a, b, c, d = s:byte(i, i + 3)
        local v = (LOOKUP[a] or 0) * 262144 + (LOOKUP[b] or 0) * 4096 + (LOOKUP[c] or 0) * 64 + (LOOKUP[d] or 0)
        local b1 = math.floor(v / 65536) % 256
        local b2 = math.floor(v / 256) % 256
        local b3 = v % 256
        bn += 1
        if c == 61 then
            buf[bn] = string.char(b1)
        elseif d == 61 then
            buf[bn] = string.char(b1, b2)
        else
            buf[bn] = string.char(b1, b2, b3)
        end
        if bn >= 2000 then
            n += 1
            out[n] = table.concat(buf, "", 1, bn)
            buf, bn = {}, 0
        end
    end
    if bn > 0 then
        n += 1
        out[n] = table.concat(buf, "", 1, bn)
    end
    return table.concat(out)
end

local function loadAsset(b64, path)
    if not (writefile and getcustomasset) then return nil end
    local ok, asset = pcall(function()
        if not (isfile and isfile(path)) then
            writefile(path, b64decode(b64))
        end
        return getcustomasset(path)
    end)
    if ok and asset and asset ~= "" then return asset end
    return nil
end

local function new(class, props, children)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do
        inst[k] = v
    end
    for _, child in ipairs(children or {}) do
        child.Parent = inst
    end
    return inst
end

local function corner(r)
    return new("UICorner", { CornerRadius = type(r) == "number" and UDim.new(0, r) or r })
end

local function stroke(color, thickness)
    return new("UIStroke", { Color = color, Thickness = thickness or 2, ApplyStrokeMode = Enum.ApplyStrokeMode.Border })
end

local function list(padding, horizontal, halign, valign)
    return new("UIListLayout", {
        Padding = UDim.new(0, padding or 0),
        FillDirection = horizontal and Enum.FillDirection.Horizontal or Enum.FillDirection.Vertical,
        HorizontalAlignment = halign or Enum.HorizontalAlignment.Left,
        VerticalAlignment = valign or Enum.VerticalAlignment.Top,
        SortOrder = Enum.SortOrder.LayoutOrder,
    })
end

local function padding(l, t, r, b)
    return new("UIPadding", {
        PaddingLeft = UDim.new(0, l or 0), PaddingTop = UDim.new(0, t or 0),
        PaddingRight = UDim.new(0, r or 0), PaddingBottom = UDim.new(0, b or 0),
    })
end

local function label(props)
    return new("TextLabel", {
        BackgroundTransparency = 1,
        Font = props.Font or FONT_BODY,
        TextSize = props.TextSize or 12,
        TextColor3 = props.Color or C.Text,
        TextXAlignment = props.Align or Enum.TextXAlignment.Left,
        TextTruncate = Enum.TextTruncate.AtEnd,
        Text = props.Text or "",
        Size = props.Size or UDim2.fromScale(1, 1),
        Position = props.Position or UDim2.new(),
        AnchorPoint = props.AnchorPoint or Vector2.new(),
        LayoutOrder = props.LayoutOrder or 0,
        AutomaticSize = props.AutomaticSize or Enum.AutomaticSize.None,
        Parent = props.Parent,
    })
end

local function box(props, fill, outline, radius, thickness)
    local class = props.Class or "Frame"
    props.Class = nil
    local f = new(class, props)
    f.BackgroundColor3 = fill or C.Panel
    f.BorderSizePixel = 0
    if radius then corner(radius).Parent = f end
    if outline then
        local s = stroke(outline, thickness or 2)
        s.Parent = f
    end
    return f
end

    local startX = math.random(math.floor(W.Width * 0.35), W.Width + 40)
    local startY = math.random(-30, math.floor(W.Height * 0.4))
    local dist = math.random(240, 380)
    local dx = dist * math.cos(math.rad(ANGLE))
    local dy = dist * math.sin(math.rad(ANGLE))
    local t = 0.8 + math.random() * 0.7

    local star = new("Frame", {
        BackgroundColor3 = C.White,
        BorderSizePixel = 0,
        AnchorPoint = Vector2.new(0.5, 0.5),
        Size = UDim2.fromOffset(len, 2),
        Position = UDim2.fromOffset(startX, startY),
        Rotation = -ANGLE,
        ZIndex = 1,
        Parent = stars,
    })
    corner(UDim.new(1, 0)).Parent = star
    new("UIGradient", { -- หัว (ซ้าย) สว่าง หาง (ขวา) จางหาย
        Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 0),
            NumberSequenceKeypoint.new(1, 1),
        }),
        Parent = star,
    })
    TweenService:Create(star, TweenInfo.new(t, Enum.EasingStyle.Linear), {
        Position = UDim2.fromOffset(startX - dx, startY + dy),
    }):Play()
    TweenService:Create(star, TweenInfo.new(t * 0.45, Enum.EasingStyle.Linear, Enum.EasingDirection.In, 0, false, t * 0.55), {
        BackgroundTransparency = 1,
    }):Play()
    task.delay(t + 0.1, function() star:Destroy() end)
end

local function fadeAndDestroy(inst, t)
    pcall(function()
        local info = TweenInfo.new(t)
        for _, d in ipairs(inst:GetDescendants()) do
            if d:IsA("TextLabel") then
                TweenService:Create(d, info, { TextTransparency = 1 }):Play()
            elseif d:IsA("UIStroke") then
                TweenService:Create(d, info, { Transparency = 1 }):Play()
            end
        end
        TweenService:Create(inst, info, { BackgroundTransparency = 1 }):Play()
    end)
    task.delay(t + 0.05, function() if inst.Parent then inst:Destroy() end end)
end

local function toast(cfg)
    if not menuState.notify then return end
    cfg = cfg or {}
    local kids = toastHolder:GetChildren()
    local count = 0
    for _, k in ipairs(kids) do if k:IsA("Frame") then count = count + 1 end end
    if count >= 4 then
        for _, k in ipairs(kids) do
            if k:IsA("Frame") then k:Destroy() break end
        end
    end
    local t = box({
        Name = "Toast", Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Y_AUTO,
        BackgroundTransparency = 0.08, ZIndex = 91, Parent = toastHolder,

local function styleTab(tab, active)
    tab.page.Visible = active
    tab.button.BackgroundTransparency = active and 0 or 1
    tab.btnStroke.Enabled = active
    tab.badge.BackgroundColor3 = active and C.White or C.SidebarAlt
    if tab.badgeLabel then tab.badgeLabel.TextColor3 = active and C.Black or C.White end
    if tab.badgeStroke then tab.badgeStroke.Enabled = not active end
end

local function newTab(cfg)
    cfg = type(cfg) == "table" and cfg or {}
    sidebarOrder = sidebarOrder + 1
    local name = tostring(cfg.Name or "Tab")
    local iconKey = ICON_BY_NAME[name] or "Star"
    local tab = setmetatable({ Name = name, Desc = cfg.Desc, iconKey = iconKey, _l = 0, _r = 0 }, TabMethods)

    tab.button = box({
        Class = "TextButton", Name = "Tab_" .. name, AutoButtonColor = false, Text = "",
        Size = UDim2.new(1, 0, 0, 30), BackgroundTransparency = 1,
        LayoutOrder = sidebarOrder, ZIndex = 3, Parent = tabHolder,
    }, C.TabActive, nil, 7)
    tab.btnStroke = stroke(C.Line, 1)
    tab.btnStroke.Enabled = false
    tab.btnStroke.Parent = tab.button
    tab.badge = iconBadge(tab.button, 4, 4, 22, ICON[iconKey] or iconKey, false, 4)
    tab.badgeLabel = tab.badge:FindFirstChildOfClass("TextLabel")
    tab.badgeStroke = tab.badge:FindFirstChildOfClass("UIStroke")
    label({
        Text = name, Font = FONT, TextSize = 12, Color = C.White,
        Position = UDim2.fromOffset(32, 0), Size = UDim2.new(1, -36, 1, 0),
        Parent = tab.button,
    }).ZIndex = 4

    tab.page = new("ScrollingFrame", {
        Name = "Page_" .. name, BackgroundTransparency = 1, BorderSizePixel = 0,
        Size = UDim2.fromScale(1, 1), Visible = false,
        ScrollBarThickness = 3, ScrollBarImageColor3 = C.White,
        ScrollingDirection = Enum.ScrollingDirection.Y,
        CanvasSize = UDim2.new(0, 0, 0, 0), ZIndex = 2, Parent = pageHost,
    })
    tab.colL = column(tab.page, 0, 10)
    tab.colR = column(tab.page, 0.5, 4)
    function tab.updateCanvas()
        tab.page.CanvasSize = UDim2.new(0, 0, 0, math.max(tab.colL.AbsoluteSize.Y, tab.colR.AbsoluteSize.Y) + 12)
    end
    tab.colL:GetPropertyChangedSignal("AbsoluteSize"):Connect(tab.updateCanvas)
    tab.colR:GetPropertyChangedSignal("AbsoluteSize"):Connect(tab.updateCanvas)

    tab.button.MouseButton1Click:Connect(function() tab:Select() end)
    allTabs[#allTabs + 1] = tab
    return tab
end

function TabMethods:Select()

    pageTitle.Text = self.Name
    pageSub.Text = self.Desc or ("NAPHUB " .. menuState.subtitle)
    applySearch()
    self.updateCanvas()
end

            TweenService:Create(track, info, { BackgroundColor3 = trackCol }):Play()
        else

        render(true)

function SectionMethods:Slider(cfg, flag)
    cfg = type(cfg) == "table" and cfg or {}
    local min, max = cfg.Minimum or 0, cfg.Maximum or 100
    local prec = cfg.Precision or 0
    local mult = 10 ^ prec
    local function snap(v)
        v = math.clamp(v, min, max)
        return math.floor(v * mult + 0.5) / mult
    end
    local value = snap(cfg.Default or min)

    local row = new("Frame", {
        Name = "Slider_" .. tostring(cfg.Name or ""), BackgroundTransparency = 1,
        Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Y_AUTO,
        LayoutOrder = self:_order(), ZIndex = 3, Parent = self._content,
    }, { list(3), padding(0, 3, 0, 3) })
    local top = new("Frame", {
        BackgroundTransparency = 1, Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Y_AUTO,
        LayoutOrder = 1, ZIndex = 3, Parent = row,
    }, { list(4, true, Enum.HorizontalAlignment.Left, Enum.VerticalAlignment.Center) })
    wlabel({
        Text = tostring(cfg.Name or ""), TextSize = 12, Color = C.Text,
        Size = UDim2.new(1, -50, 0, 0), LayoutOrder = 1, Parent = top,
    }).ZIndex = 4
    local valLbl = label({
        Text = "", Font = FONT, TextSize = 11, Color = C.SubText,
        Align = Enum.TextXAlignment.Right,
        Size = UDim2.fromOffset(46, 16), LayoutOrder = 2, Parent = top,
    })
    valLbl.ZIndex = 4
    local holder = new("Frame", {
        BackgroundTransparency = 1, Active = true,
        Size = UDim2.new(1, 0, 0, 20), LayoutOrder = 2, ZIndex = 4, Parent = row,
    })
    local bar = box({
        AnchorPoint = Vector2.new(0, 0.5), Position = UDim2.new(0, 0, 0.5, 0),
        Size = UDim2.new(1, 0, 0, 8), ZIndex = 4, Parent = holder,
    }, C.Track, C.Outline, UDim.new(1, 0), 1)
    local fill = box({
        Size = UDim2.fromScale(0, 1), ZIndex = 5, Parent = bar,
    }, C.White, nil, UDim.new(1, 0))
    local knob = box({
        AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0, 0.5),
        Size = UDim2.fromOffset(14, 14), ZIndex = 6, Parent = bar,
    }, C.White, C.Black, UDim.new(1, 0), 2)

    local function render()
        local a = (max > min) and ((value - min) / (max - min)) or 0
        fill.Size = UDim2.fromScale(a, 1)
        knob.Position = UDim2.fromScale(a, 0.5)
        valLbl.Text = string.format("%." .. prec .. "f", value)
    end
    local function setValue(v, fire)
        v = snap(v)
        if v ~= value then
            value = v
            render()
            if fire then safeCall(cfg.Callback, value) end
        end
    end
    local function fromX(x)
        local w = bar.AbsoluteSize.X
        if w <= 0 then return end
        local a = math.clamp((x - bar.AbsolutePosition.X) / w, 0, 1)
        setValue(min + (max - min) * a, true)
    end
    local dragging = false
    holder.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            fromX(input.Position.X)
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch) then
            fromX(input.Position.X)
        end
    end)
    render()
    registerRow(row, cfg.Name)
    if flag then
        flagRegistry[flag] = {
            get = function() return value end,
            set = function(v) if type(v) == "number" then setValue(v, true) end end,
        }
    end
    return {
        Set = function(_, v) setValue(v, true) end,
        Get = function() return value end,
    }
end

----------------------------------------------------------------
-- Config (บันทึก/โหลดค่าสวิตช์-สไลเดอร์ ลงโฟลเดอร์ workspace/NAPHUB/settings)
----------------------------------------------------------------
local function cfgDir() return configFolder .. "/settings" end

local function ensureDirs()
    if not (makefolder and isfolder) then return end
    pcall(function()
        if not isfolder(configFolder) then makefolder(configFolder) end
        if not isfolder(cfgDir()) then makefolder(cfgDir()) end
    end)
end

local function listConfigs()
    local out = {}
    if listfiles and isfolder and isfolder(cfgDir()) then
        local ok, files = pcall(listfiles, cfgDir())
        if ok and type(files) == "table" then
            for _, f in ipairs(files) do
                local n = tostring(f):match("([^/\\]+)%.json$")
                if n then out[#out + 1] = n end
            end
        end
    end
    table.sort(out)
    return out
end

local function saveConfig(name)
    name = tostring(name or ""):gsub('[/\\:*?"<>|]', ""):gsub("^%s+", ""):gsub("%s+$", "")
    if name == "" then return false, "ยังไม่ได้ตั้งชื่อ Config" end
    if not writefile then return false, "Executor ไม่รองรับการเขียนไฟล์" end
    ensureDirs()
    local data = {}
    for flag, h in pairs(flagRegistry) do
        local ok, v = pcall(h.get)
        if ok then data[flag] = v end
    end
    local ok, err = pcall(function()
        writefile(cfgDir() .. "/" .. name .. ".json", HttpService:JSONEncode(data))
    end)
    if not ok then return false, tostring(err) end
    return true, name
end

local function loadConfig(name)
    if not (readfile and isfile) then return false, "Executor ไม่รองรับการอ่านไฟล์" end
    local path = cfgDir() .. "/" .. tostring(name) .. ".json"
    if not isfile(path) then return false, "ไม่พบไฟล์ Config" end
    local ok, data = pcall(function() return HttpService:JSONDecode(readfile(path)) end)
    if not ok or type(data) ~= "table" then return false, "ไฟล์ Config เสีย" end
    for flag, v in pairs(data) do
        local h = flagRegistry[flag]
        if h then pcall(h.set, v) end
    end
    return true
end

local function startLoop(flag, fn, delayKey)
	Run[flag] = (Run[flag] or 0) + 1
	local id = Run[flag]
	task.spawn(function()
		while State[flag] and Run[flag] == id do
			local ok, err = pcall(fn)

			task.wait(Cfg[delayKey])
		end
	end)
end

end

-- วาร์ปไปหน้าเส้น แล้วเช็กซ้ำว่าถึงจริง (คืน true ถ้าอยู่ที่เส้นแล้ว)
-- จุดวาร์ป = ถอยหลังจากตำแหน่งหน้าเส้นตามระยะ LineOffset (ใกล้เส้น แต่ไม่เข้าเส้นขว้าง)
local function throwTarget()
	if not Cfg.ThrowCF then return nil end
	return Cfg.ThrowCF * CFrame.new(0, 0, Cfg.LineOffset or 0) -- +Z ในตัวละคร = ด้านหลัง
end

local function tpThrowLine()
	local target = throwTarget()
	if not Cfg.TPThrow or not target then return true end
	for _ = 1, 3 do
		local hrp = getHRP()
		if not hrp then return false end
		if (hrp.Position - target.Position).Magnitude <= 2 then return true end
		hrp.CFrame = target
		hrp.AssemblyLinearVelocity = Vector3.zero
		hrp.AssemblyAngularVelocity = Vector3.zero
		task.wait(0.1)
	end
	local hrp = getHRP()
	return hrp ~= nil and (hrp.Position - target.Position).Magnitude <= 4
end

-- ===== ปรับค่าเพอร์เฟกต์อัตโนมัติ: ถ้าไม่ได้ PERFECT จะลองค่าถัดไปจนเจอแล้วล็อกไว้ =====
local SWEEP = { 0.99, 1.0, 0.97, 0.96, 0.98, 0.95 }
local tune = { idx = 0, locked = false, fails = 0, value = nil }

local function baseValue()
	return tune.value or Cfg.PerfectValue
end

local function afterResult(res)
	if not Cfg.AutoTune or not res then return end
	if res == "PERFECT" then
		tune.fails = 0
		if not tune.locked then
			tune.locked = true
			Cfg.PerfectValue = baseValue()
			tune.value = Cfg.PerfectValue
			notify(string.format("เจอค่าเพอร์เฟกต์ %.3f ล็อกไว้แล้ว", Cfg.PerfectValue))
		end
	else
		if tune.locked then
			tune.fails = tune.fails + 1
			if tune.fails >= 3 then tune.locked = false; tune.idx = 0; tune.fails = 0 end
		else
			tune.idx = tune.idx % #SWEEP + 1
			tune.value = SWEEP[tune.idx]
		end
	end
end

-- จับตำแหน่งหน้าเส้น + ค่าที่คุณกดเอง: ถ้าขว้างเองแล้วได้ PERFECT จะจำค่านั้นไว้ใช้ต่อ
local captureInstalled = false
local function installCapture()
	if captureInstalled or not (hookmetamethod and getnamecallmethod) then return end
	captureInstalled = true
	local mk = newcclosure or function(f) return f end
	local chk = checkcaller or function() return false end
	local old
	old = hookmetamethod(game, "__namecall", mk(function(self, ...)
		if getnamecallmethod() == "InvokeServer" and not chk() then
			local n = typeof(self) == "Instance" and self.Name
			if n == "Throw_Start" then
				task.defer(function()
					local hrp = getHRP()
					if hrp then
						Cfg.ThrowCF = hrp.CFrame
						saveLineFile()
					end
				end)
			elseif n == "Throw_Resolve" then
				local v = tonumber((...))
				task.defer(function()
					task.wait(0.4)
					if v and readThrowResult() == "PERFECT" then
						Cfg.PerfectValue = v
						tune.value, tune.locked, tune.fails = v, true, 0
						notify(string.format("จำค่า Perfect จากที่คุณกดแล้ว: %.4f", v))
					end
				end)
			end
		end
		return old(self, ...)
	end))
end
installCapture()

			local id = m:GetAttribute("AnimalId")
			if id then incById[id] = inc end
			local key = tostring(m:GetAttribute("AnimalName") or m.Name) .. "|" .. tostring(m:GetAttribute("Mutation") or "")
			if inc > (incByKey[key] or 0) then incByKey[key] = inc end
		end
	end

	local list = {}
	for _, f in ipairs(animals:GetChildren()) do
		local inc = incById[f.Name]
		if not inc then
			local mn, mu = f:FindFirstChild("ModelName"), f:FindFirstChild("Mutation")
			inc = incByKey[tostring(mn and mn.Value or "") .. "|" .. tostring(mu and mu.Value or "")]
		end
		table.insert(list, { id = f.Name, inc = inc })
	end
	table.sort(list, function(x, y) return (x.inc or -1) > (y.inc or -1) end)

	local sold, unknown = 0, 0
	for i = Cfg.KeepTop + 1, #list do
		local e = list[i]
		if e.inc then -- ขายเฉพาะตัวที่รู้ Income แน่ชัด
			local ok, res = invoke("Plot_SellAnimal", e.id)
			if ok and res ~= false then sold = sold + 1 end
			task.wait(0.15)
		else
			unknown = unknown + 1
		end
	end
	invoke("Plot_EquipBest")
	notify("ขายแล้ว " .. sold .. " ตัว (เก็บ " .. math.min(Cfg.KeepTop, #list) .. " ตัวบนสุด" .. (unknown > 0 and (", ข้าม " .. unknown .. " ตัวที่ไม่รู้ Income") or "") .. ")")
end

-- ESP สัตว์ของผู้เล่นอื่น (ชื่อ | Mutation | Income)
local espFolder
local function clearESP()
	if espFolder then espFolder:Destroy(); espFolder = nil end
end

local function doESP()
	clearESP()
	espFolder = Instance.new("Folder")
	espFolder.Name = "NAPHUB_ESP"
	espFolder.Parent = guiRoot()
	local plots = workspace:FindFirstChild("Plots")
	if not plots then return end
	for _, plot in ipairs(plots:GetChildren()) do
		local owner = plot:GetAttribute("OwnerUserId")
		local animals = plot:FindFirstChild("Animals")
		if animals and owner and owner ~= LP.UserId then
			for _, a in ipairs(animals:GetChildren()) do
				local part = a:FindFirstChild("RootPart") or a.PrimaryPart or a:FindFirstChildWhichIsA("BasePart", true)
				if part then
					local bb = Instance.new("BillboardGui")
					bb.Adornee = part
					bb.AlwaysOnTop = true
					bb.Size = UDim2.fromOffset(180, 34)
					bb.StudsOffset = Vector3.new(0, 6, 0)
					local t = Instance.new("TextLabel")
					t.Size = UDim2.fromScale(1, 1)
					t.BackgroundTransparency = 1
					t.TextColor3 = Color3.fromRGB(255, 255, 255)
					t.TextStrokeTransparency = 0.3
					t.Font = Enum.Font.GothamBold
					t.TextSize = 13
					t.TextWrapped = true
					t.Text = string.format("%s | %s | %s/s",
						tostring(a:GetAttribute("DisplayName") or a.Name),
						tostring(a:GetAttribute("Mutation") or "-"),
						fmt(a:GetAttribute("Income")))
					t.Parent = bb
					bb.Parent = espFolder
				end
			end
		end
	end
end
